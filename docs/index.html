atlas-ragnarok

Tech-blue thunder above. Crimson fire below. Pure black through the middle. 

// atlas-ragnarok — the palette in action

const storm = forge({
  background: '#000000',
  thunder: '#3b82f6',  // top edges
  ember:   '#ff8080',  // bottom edges
  intensity: 0.95,
});

if (storm.isStable()) {
  deploy(storm);
} else {
  throw new Error('palette diverged');
}
